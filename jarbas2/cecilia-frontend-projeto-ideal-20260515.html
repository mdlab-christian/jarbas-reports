1. Visão do produto #

Hoje o operador da Cecília não responde rapidamente perguntas básicas — "qual OAB está rodando agora?", "quando foi o último ciclo de SP535101?", "quantos processos estão na fila?". A informação existe no banco mas está fragmentada em 7 sub-tabs sem foco operacional, há 1 card morto no header (Worker B fantasma) e o cron está em 3 lugares diferentes (cron_hours global, schedules_json por-OAB, cecilia_capture_config.horarios).

A v2 unifica tudo num console operacional:

🎯 Painel de Cron unificado

SSOT = cecilia_config.schedules_json. Cron por OAB, sem fonte global conflitante. Modo: horário fixo ou intervalo. Toggle ON/OFF por OAB.

D-04 · D-17 · D-23

📊 Status em tempo real

5 OABs com pill verde/amarelo/vermelho, último ciclo relativo, próximo ciclo agendado, intimações/24h. Realtime via cecilia_health reusando canal de useCeciliaHealth.

D-02 · D-18

📈 Timeline 7 dias

Heatmap 5 OABs × 7 dias via Recharts AreaChart consistente com CustoSparkline. RPC rpc_cecilia_timeline_7d agrega cecilia_ciclos direto (76 linhas reais).

D-08 · D-15 · D-21

📋 Fila operacional

cecilia_processos_fila com 180 linhas. Reusa useCeciliaHealth (sem hook novo, sem polling). KPI cards + tabela filtrada.

D-09 · D-13

📤 Upload XLS

Drag & drop + Edge Function cecilia-xls-processor + Drawer pós-upload com progress + CTA "Ir para Fila". Histórico dos últimos 10 uploads.

D-10

⚖️ Intimações classificadas

1452 intimações fonte='cecilia_v2'. Filtros urgência/tipo_acao/validacao. Bulk actions só para auto_aprovada + pendente_classificacao. Drawer detalhe com análise IA + raw.

D-11
✨ Princípio orientador: reaproveitar, não recriar
Componentes que JÁ existem e serão extraídos para shared em vez de recriados: PauseWorldDialog (CeciliaSubComandos:52-137), StatusPill/StatusDot (2 implementações duplicadas), CronEditor (extender para por-OAB), useCeciliaHealth (alimentar tab Fila direto), Recharts AreaChart padrão.

-- supabase/migrations/20260515_rpc_cecilia_status_oabs.sql
CREATE OR REPLACE FUNCTION public.rpc_cecilia_status_oabs_v2(
  p_organizacao_id UUID
)
RETURNS TABLE (
  oab TEXT,
  tribunal TEXT,
  ativo BOOLEAN,
  cron_ativo BOOLEAN,
  modo_padrao TEXT,
  horarios_ssot JSONB,        -- de cecilia_config.schedules_json (D-17)
  ultimo_ciclo_at TIMESTAMPTZ,
  proximo_ciclo_at TIMESTAMPTZ,
  taxa_sucesso_24h NUMERIC,
  intimacoes_24h INTEGER,
  alarmes_24h INTEGER,
  login_falhas_consec INTEGER
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_org UUID := p_organizacao_id;
BEGIN
  RETURN QUERY
  SELECT
    ja.oab,
    ja.tribunal,
    ja.ativo,
    (cc.schedules_json ? ja.oab) AS cron_ativo,
    ccc.modo_padrao,
    cc.schedules_json -> ja.oab AS horarios_ssot,
    ch.ultimo_ciclo_at,
    ch.proximo_ciclo_at,
    ch.taxa_sucesso_24h,
    (SELECT COUNT(*)::INTEGER FROM intimacoes i
       WHERE i.organizacao_id = v_org
         AND i.fonte = 'cecilia_v2'
         AND i.oab = ja.oab
         AND i.created_at >= NOW() - INTERVAL '24 hours')::INTEGER,
    (SELECT COUNT(*)::INTEGER FROM cecilia_alarmes a
       WHERE a.organizacao_id = v_org
         AND a.oab = ja.oab
         AND a.resolvido_at IS NULL
         AND a.created_at >= NOW() - INTERVAL '24 hours')::INTEGER,
    ja.login_falhas_consecutivas
  FROM jarbas_accounts ja
  LEFT JOIN cecilia_health ch
    ON ch.oab = ja.oab AND ch.worker = 'A'
  LEFT JOIN cecilia_capture_config ccc
    ON ccc.oab = ja.oab AND ccc.organizacao_id = v_org
  LEFT JOIN cecilia_config cc
    ON cc.organizacao_id = v_org
  WHERE ja.organizacao_id = v_org
  ORDER BY ja.oab;
END $$;

GRANT EXECUTE ON FUNCTION public.rpc_cecilia_status_oabs_v2 TO authenticated;
M3 RPC · rpc_cecilia_timeline_7d
P1 ~10min agrega cecilia_ciclos 
-- supabase/migrations/20260515_rpc_cecilia_timeline_7d.sql
CREATE OR REPLACE FUNCTION public.rpc_cecilia_timeline_7d(
  p_organizacao_id UUID
)
RETURNS TABLE (
  oab TEXT,
  data DATE,
  total_ciclos INTEGER,
  ciclos_ok INTEGER,
  ciclos_erro INTEGER,
  ciclos_pausado INTEGER,
  intimacoes_dia INTEGER
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  RETURN QUERY
  WITH dias AS (
    SELECT generate_series(
      (CURRENT_DATE - INTERVAL '6 days')::DATE,
      CURRENT_DATE::DATE,
      '1 day'::INTERVAL
    )::DATE AS data
  ),
  oabs AS (
    SELECT DISTINCT ja.oab FROM jarbas_accounts ja
    WHERE ja.organizacao_id = p_organizacao_id
  ),
  matriz AS (
    SELECT o.oab, d.data FROM oabs o CROSS JOIN dias d
  )
  SELECT
    m.oab,
    m.data,
    COALESCE(COUNT(cc.id)::INTEGER, 0) AS total_ciclos,
    COALESCE(COUNT(cc.id) FILTER (WHERE cc.status = 'ok')::INTEGER, 0) AS ciclos_ok,
    COALESCE(COUNT(cc.id) FILTER (WHERE cc.status = 'erro')::INTEGER, 0) AS ciclos_erro,
    COALESCE(COUNT(cc.id) FILTER (WHERE cc.status = 'pausado')::INTEGER, 0) AS ciclos_pausado,
    COALESCE((SELECT COUNT(*)::INTEGER FROM intimacoes i
                WHERE i.organizacao_id = p_organizacao_id
                  AND i.fonte = 'cecilia_v2'
                  AND i.oab = m.oab
                  AND i.created_at::DATE = m.data), 0)
  FROM matriz m
  LEFT JOIN cecilia_ciclos cc
    ON cc.oab = m.oab
   AND cc.organizacao_id = p_organizacao_id
   AND cc.iniciado_em::DATE = m.data
  GROUP BY m.oab, m.data
  ORDER BY m.oab, m.data;
END $$;

GRANT EXECUTE ON FUNCTION public.rpc_cecilia_timeline_7d TO authenticated;
