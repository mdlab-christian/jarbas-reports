Query principal

// useLeadsCrm.ts — dados do Kanban
const { data, isLoading } = useQuery({
  queryKey: ['crm-leads', orgId],
  queryFn: async () => {
    const { data, error } = await supabase
      .from('view_crm_leads_kanban')
      .select('*')
      .eq('organizacao_id', orgId)
      .in('coluna_crm', [
        'novos', 'em_atendimento', 'aguardando_documentos',
        'aguardando_assinatura', 'triagem', 'prontos'
      ])
      .order('updated_at', { ascending: false })
      .limit(1000)
    if (error) throw error
    return data
  },
  staleTime: 30_000,
  refetchInterval: 60_000, // fallback se Realtime cair
})

handleDragEnd — atualização de coluna

const handleDragEnd = async ({ active, over }) => {
  if (!over || active.id === over.id) return

  const novaColuna = over.data.current?.coluna_crm
  if (!novaColuna) return

  // Atualiza diretamente no Supabase
  await supabase
    .from('leads')
    .update({ coluna_crm: novaColuna })
    .eq('id', active.id)

  // Invalida query para refetch
  queryClient.invalidateQueries({ queryKey: ['crm-leads', orgId] })
}

Schema da tabela restricoes

CREATE TABLE restricoes (
  id                UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  organizacao_id    UUID NOT NULL REFERENCES organizacoes(id) ON DELETE CASCADE,
  lead_id           UUID REFERENCES leads(id),
  cliente_id        UUID REFERENCES clientes(id),

  empresa_id        UUID REFERENCES empresas(id),
  empresa_nome      TEXT,                       -- denormalizado para histórico
  orgao_nome        TEXT,                       -- SCPC/Boa Vista | Serasa | SPC | Quod | BCB
  tipo              TEXT CHECK (tipo IN ('comprador', 'devedor', 'fiador', 'avalista')),

  codigo_contrato   TEXT,
  data_restricao    DATE,
  valor             INT8,                       -- centavos (INT8, não DECIMAL)

  processavel       BOOLEAN DEFAULT TRUE,      -- herdado da empresa se do catálogo
  processo_id       UUID REFERENCES processos(id),  -- NULL = sem processo vinculado

  deleted_at        TIMESTAMPTZ,               -- soft delete obrigatório
  created_at        TIMESTAMPTZ DEFAULT now(),
  updated_at        TIMESTAMPTZ DEFAULT now()
);

Arquitetura dos 3 Tiers

Tier 1 — Central (documentos gerais do lead) categoria = documento_lead

Filtro: restricao_id IS NULL AND categoria = 'documento_lead'

Tipos aceitos: RG, CPF, CNH, Comprovante Residência, Comprovante Renda, Procuração, Contrato, Extrato Bancário, Outros

Upload: drag-and-drop na área central

Tier 2 — Pasta por restrição (documentos da restrição) categoria = documento_restricao

Filtro: restricao_id = {id} AND categoria = 'documento_restricao'

UI: pasta colapsável por empresa. Upload dentro da pasta da restrição específica.

Tier 3 — Gerado por n8n useGerarContrato

Origem: gerado pelo hook useGerarContrato via webhook n8n

Tipos: contratos gerados, procurações emitidas, documentos Midas

Inferência de tipo por nome de arquivo

function inferirTipoDocumento(nome: string): TipoDocumento {
  const n = nome.toLowerCase()
  if (/rg/.test(n))                   return 'rg'
  if (/cpf/.test(n))                  return 'cpf'
  if (/cnh/.test(n))                  return 'cnh'
  if (/comprovante.*resid/.test(n))  return 'comprovante_residencia'
  if (/comprovante.*renda/.test(n))  return 'comprovante_renda'
  if (/procuracao/.test(n))          return 'procuracao'
  if (/contrato/.test(n))           return 'contrato'
  if (/extrato/.test(n))            return 'extrato_bancario'
  return 'outro'
}

view_crm_leads_kanban — SQL completo

CREATE OR REPLACE VIEW view_crm_leads_kanban AS
SELECT
  l.id, l.organizacao_id, l.nome, l.cpf, l.data_nascimento,
  l.whatsapp, l.coluna_crm, l.atendente_id,

  -- Atendente
  u.nome AS atendente_nome,

  -- Advogado
  l.advogado_id,
  adv.nome AS advogado_nome,
  adv.oab,
  adv.oab_uf,

  -- Origem
  l.origem_id,
  lo.nome AS origem_nome,

  -- Estado/UF
  l.estado_id,
  ed.uf AS estado_sigla,

  -- Metadados
  l.awaiting_human, l.ultimo_contato, l.score,
  l.etiquetas, l.updated_at, l.created_at,

  -- Contadores de restrições (subquery — NÃO JSONB denormalizado)
  (SELECT COUNT(*) FROM restricoes r
   WHERE r.lead_id = l.id AND r.deleted_at IS NULL
  ) AS restricoes_count,

  (SELECT COUNT(*) FROM restricoes r
   WHERE r.lead_id = l.id
     AND r.processavel = true
     AND r.processo_id IS NULL
     AND r.deleted_at IS NULL
  ) AS restricoes_processaveis_count,

  -- Documentos pendentes (ainda via JSONB — gap a resolver)
  (SELECT COUNT(*) FROM jsonb_array_elements(COALESCE(l.documentos, '[]'::jsonb)) d
   WHERE (d.value ->> 'situacao') = 'pendente'
  ) AS documentos_pendentes_count,

  l.indicacao_id,

  -- Notas via LATERAL (eficiente)
  COALESCE(nota_agg.notas_count, 0) AS notas_count,
  nota_agg.ultima_nota

FROM leads l
LEFT JOIN users u           ON l.atendente_id = u.id
LEFT JOIN advogados adv     ON l.advogado_id = adv.id
LEFT JOIN leads_origens lo  ON l.origem_id = lo.id
LEFT JOIN estados_distribuicao ed ON l.estado_id = ed.id
LEFT JOIN LATERAL (
  SELECT
    COUNT(*) AS notas_count,
    (SELECT la2.descricao
     FROM leads_atividades la2
     WHERE la2.lead_id = l.id AND la2.tipo = 'nota'
     ORDER BY la2.created_at DESC LIMIT 1
    ) AS ultima_nota
  FROM leads_atividades la
  WHERE la.lead_id = l.id AND la.tipo = 'nota'
) nota_agg ON TRUE

WHERE l.deleted_at IS NULL

// Subscription em leads para refetch automático
useEffect(() => {
  const channel = supabase
    .channel(`crm-leads-${orgId}`)
    .on(
      'postgres_changes',
      { event: '*', schema: 'public', table: 'leads',

// NUNCA decidir no frontend baseado em response.success
// ERRADO:
if (response.success) navigate('/processos')

// SEMPRE usar switch no action
switch (response.action) {
  case 'toast_refresh':
    toast.success(response.message)
    queryClient.invalidateQueries({ queryKey: ['crm-leads', orgId] })
    break
  case 'toast_error':
    toast.error(response.message)
    break
  case 'redirect':
    navigate(response.data.url)
    break
  case 'close_modal':
    setModalOpen(false)
    break
  case 'webhook_triggered':
    toast.info(response.message)
    break
  case 'noop': break
}

Gaps & Pendências

8 abertos
!
Gaps conhecidos — não bloqueia produção atual

Todos os itens abaixo são problemas ou funcionalidades incompletas identificadas. Classificados por prioridade P0/P1/P2.

P1 · GAP-01

WH-CRM-02 — Follow-up automático em lote não implementado

O crmWebhookService possui a função mas o workflow n8n não existe. Follow-up em lote via WhatsApp/email está bloqueado até criação do workflow.

P1 · GAP-02

WH-CRM-05 — Envio WhatsApp individual bloqueado

Service implementado mas requer MDZap em produção plena. Webhook n8n MDZAP-ENVIAR-MENSAGEM existe mas a integração CRM→MdZap individual não está ativa.

P2 · GAP-03

WH-CRM-03 — Notificações pós-triagem desativadas

O webhook foi desativado pois a RPC rpc_crm_aprovar_triagem executa o core. Pasta Drive e notificações pós-triagem ficam para sprint futuro.

P1 · GAP-04

Busca Kanban client-side — performance com volumes altos

A busca é aplicada localmente sobre os 1000 leads carregados em memória. Com mais de 500 leads ativos, pode haver lentidão perceptível no filtro. Busca server-side não implementada para o Kanban.

P1 · GAP-05

Documentos JSONB — campo leads.documentos ainda denormalizado

Parte dos documentos ainda vive em leads.documentos JSONB. Não existe tabela documentos_lead separada com todos os campos. O contador de documentos_pendentes_count na view depende de jsonb_array_elements — estrutura mais frágil.

P2 · GAP-06

MdZap score sync — awaiting_human sem sincronização bidirecional

O campo awaiting_human é atualizado pelo MDZap mas não há sync automático reverso. Webhooks WH-CRM-04/06/07/08 estão reservados mas não implementados.

P2 · GAP-07

Aba Radar — não documentada em detalhe

A aba Radar está implementada e funcional mas não tem documentação técnica nesta referência. Consultar spec e3__crm para detalhes.

P2 · GAP-08

FRONT-010 MdZap — foto de perfil (photo_url) não implementada

O campo photo_url não é capturado no webhook de entrada do WhatsApp. A foto de perfil do lead/contato MdZap não é exibida no drawer nem no card.
