Q-07 REVISADO — Trigger PostgreSQL
Antes
Trigger em status = 'ag_preparacao'
Depois (correto)
Trigger em etapa_distribuicao = 'preparacao'
CREATE OR REPLACE FUNCTION notify_davi_preparacao() RETURNS TRIGGER AS $$ BEGIN IF NEW.etapa_distribuicao = 'preparacao' AND (OLD.etapa_distribuicao IS DISTINCT FROM 'preparacao') THEN PERFORM net.http_post( url := 'https://qdivfairxhdihaqqypgb.supabase.co/functions/v1/davi-validar', headers := '{"Authorization": "Bearer " || current_setting(''app.service_role_key'')}'::jsonb, body := json_build_object('processo_id', NEW.id, 'organizacao_id', NEW.organizacao_id)::text ); END IF; RETURN NEW; END; $$ LANGUAGE plpgsql SECURITY DEFINER; CREATE TRIGGER davi_preparacao_trigger AFTER UPDATE ON processos FOR EACH ROW EXECUTE FUNCTION notify_davi_preparacao();
Slot 3 REVISADO — Identidade vs CNH
Antes
GPT com score de confiança; se < 80% → Revisão Humana
Agora (correto)
Identificar pelo nome do arquivo — o nome já diz (3- Identidade ou 3- CNH). GPT confirma. SEMPRE passa — não manda para Revisão Humana.
Por que
Se o arquivo está nomeado errado, isso seria erro do atendente de muito fácil detecção. O nome do arquivo é fonte confiável para este slot específico.
Checklist atualizado — Slot 3
  • Buscar em processos_documentos onde tipo_documento = 'identidade' E processo_id = X
  • Ler nome do arquivo: contém "CNH" → tipo = CNH; contém "Identidade" ou "RG" → tipo = Identidade
  • Registrar tipo_identificado para Mia usar no EPROC
  • Sempre aprovado (não vai para Revisão Humana neste slot)

Como o Davi Acessa os Documentos — Fluxo Técnico

Slot 1 — Petição Inicial (PI)
-- Verificar se PI foi gerada SELECT peticao_status, template_peticao_id, peticao_gerada_em FROM processos WHERE id = :processo_id; -- Verificar metadado do template SELECT nome, categoria_empresa FROM templates_peticao -- confirmar nome real da tabela WHERE id = :template_peticao_id; -- Verificar empresa SELECT categoria FROM empresas WHERE id = :empresa_id;
Slot 2 — Procuração
-- URL direto na tabela processos SELECT procuracao_url FROM processos WHERE id = :processo_id; -- OU via processos_documentos SELECT arquivo_url, storage_path FROM processos_documentos WHERE processo_id = :processo_id AND tipo_documento = 'procuracao' AND deleted_at IS NULL;
Slot 3 — Identidade / CNH
SELECT nome, arquivo_url, storage_path FROM processos_documentos WHERE processo_id = :processo_id AND tipo_documento = 'identidade' AND deleted_at IS NULL; -- Logica no codigo: se nome ILIKE '%CNH%' -> tipo_cnh = true, else tipo_identidade = true
Slot 4 — Comprovante de Residência
SELECT nome, arquivo_url, storage_path FROM processos_documentos WHERE processo_id = :processo_id AND tipo_documento = 'residencia' AND deleted_at IS NULL; -- Enviar PDF para GPT-4.1 Mini para extrair empresa emissora -- Comparar com empresas.nome WHERE id = processos.empresa_id
Slot 5a — Renda / CTPS
SELECT nome, arquivo_url FROM processos_documentos WHERE processo_id = :processo_id AND tipo_documento = 'renda_ctps' AND deleted_at IS NULL; -- Verificar apenas presenca
Slot 5b — Situação Cadastral
-- QUANDO tipo_documento = 'situacao_cadastral' for criado: SELECT nome, arquivo_url FROM processos_documentos WHERE processo_id = :processo_id AND tipo_documento = 'situacao_cadastral' AND deleted_at IS NULL; -- Ate la: buscar por nome do arquivo (case-insensitive) SELECT nome, arquivo_url FROM processos_documentos WHERE processo_id = :processo_id AND LOWER(nome) LIKE '%situa%cadastral%' AND deleted_at IS NULL;
Slot 6 — Comprovante de Restrição
-- URL direto (para processo nao-BV) SELECT comprovante_restricao_url, comprovante_restricao_status FROM processos WHERE id = :processo_id; -- Verificar restricao associada SELECT r.empresa_id, r.empresa_nome, r.codigo_contrato, r.valor, r.processavel FROM restricoes r WHERE r.processo_id = :processo_id AND r.deleted_at IS NULL; -- Para processo BV: empresa_id = '4ff6b763-646e-4788-9e33-20e2a7763d8e' -> usar auditoria_url
