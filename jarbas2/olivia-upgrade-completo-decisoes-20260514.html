function useProcessTimer(historicoId: string, isRunning: boolean) { const [accumulatedMs, setAccumulatedMs] = useState(0); const startRef = useRef<number | null>(null); const visibleRef = useRef(document.visibilityState === 'visible'); useEffect(() => { if (!isRunning) { startRef.current = null; return; } const tick = () => { if (visibleRef.current && startRef.current) { setAccumulatedMs(prev => prev + (Date.now() - startRef.current!)); startRef.current = Date.now(); } }; const interval = setInterval(tick, 1000); const onVisibility = () => { visibleRef.current = document.visibilityState === 'visible'; if (visibleRef.current) startRef.current = Date.now(); else { tick(); startRef.current = null; } }; document.addEventListener('visibilitychange', onVisibility); if (visibleRef.current) startRef.current = Date.now(); return () => { clearInterval(interval); document.removeEventListener('visibilitychange', onVisibility); }; }, [isRunning]); // Persist debounced 5s useDebouncedEffect(() => { supabase.from('olivia_historico') .update({ tempo_processamento_ms: accumulatedMs }) .eq('id', historicoId); }, [accumulatedMs], 5000); return formatMs(accumulatedMs); // "02:34" }
